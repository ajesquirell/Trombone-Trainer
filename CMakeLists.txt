cmake_minimum_required(VERSION 3.24)
project(TromboneTrainer VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_OBJCXX_STANDARD 20)
set(CMAKE_OBJCXX_STANDARD_REQUIRED ON)
set(CMAKE_OBJCXX_EXTENSIONS OFF)
option(TRAINER_BUILD_APP "Build Lomse compatibility test and trainer" ON)
option(TRAINER_BUILD_BOOTSTRAP "Build PGE3-only platform diagnostic" OFF)
include(CTest)
if(BUILD_TESTING AND NOT EMSCRIPTEN)
  add_executable(core_tests tests/core_tests.cpp)
  target_include_directories(core_tests PRIVATE src)
  add_test(NAME core COMMAND core_tests)
endif()
if(TRAINER_BUILD_APP OR TRAINER_BUILD_BOOTSTRAP)
  include(cmake/PGE3.cmake)
  if(BUILD_TESTING AND NOT EMSCRIPTEN)
    add_test(NAME pge3_font_payload COMMAND ${CMAKE_COMMAND}
      "-DUPSTREAM_HEADER=${pge3_source_SOURCE_DIR}/olcPixelGameEngine3.h"
      "-DGENERATED_HEADER=${CMAKE_CURRENT_BINARY_DIR}/pge3/olcPixelGameEngine3.h"
      -P "${CMAKE_CURRENT_SOURCE_DIR}/tests/pge3_font_test.cmake")
  endif()
  if(NOT EMSCRIPTEN)
    add_executable(trombone-preview EXCLUDE_FROM_ALL tools/trombone_preview.cpp)
    target_include_directories(trombone-preview PRIVATE src)
    trainer_platform(trombone-preview)
  endif()
endif()
if(TRAINER_BUILD_BOOTSTRAP)
  add_executable(pge3-bootstrap src/bootstrap.cpp)
  trainer_platform(pge3-bootstrap)
endif()
if(TRAINER_BUILD_APP)
  include(cmake/Lomse.cmake)
  add_library(notation STATIC src/Notation.cpp)
  if(APPLE)
    set_source_files_properties(src/Notation.cpp PROPERTIES LANGUAGE OBJCXX)
  endif()
  target_link_libraries(notation PUBLIC pge3 lomse)
  target_include_directories(notation PUBLIC src "${lomse_SOURCE_DIR}/include" "${lomse_BINARY_DIR}")
  target_compile_definitions(notation PRIVATE TRAINER_FONT_DIR="${TRAINER_FONT_DIR}/")
  foreach(app IN ITEMS notation-check trombone-trainer)
    if(app STREQUAL "notation-check")
      add_executable(${app} src/compatibility.cpp)
    else()
      add_executable(${app} src/main.cpp src/Game.cpp)
    endif()
    target_link_libraries(${app} PRIVATE notation)
    trainer_platform(${app})
    if(EMSCRIPTEN)
      target_link_options(${app} PRIVATE -sUSE_FREETYPE=1 -sUSE_ZLIB=1
        "--preload-file" "${TRAINER_ASSET_DIR}@/assets")
    endif()
  endforeach()
  if(EMSCRIPTEN)
    add_custom_target(web-site
      COMMAND ${CMAKE_COMMAND}
        "-DINPUT_DIR=$<TARGET_FILE_DIR:trombone-trainer>"
        "-DOUTPUT_DIR=${CMAKE_CURRENT_BINARY_DIR}/site"
        -P "${CMAKE_CURRENT_SOURCE_DIR}/cmake/StageWeb.cmake"
      DEPENDS trombone-trainer
      VERBATIM)
  endif()
endif()
